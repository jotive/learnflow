# LearnFlow Backend v1 Implementation Plan

> **For agentic workers:** Implementar tarea por tarea. Steps usan checkbox (`- [ ]`) para tracking.

**Goal:** FastAPI backend for LearnFlow — learning-path tracker with compliance gating, JWT auth, assignment + fake notification, ≥75% pytest coverage, Docker.

**Architecture:** Clean/hexagonal, three layers. `app/domain` (Pydantic entities, enums, exceptions, ports, policies — zero framework imports) ← `app/application` (one use case class per business action, constructor-injected ports) ← `app/infrastructure` (SQLAlchemy repos, FastAPI routers, JWT, console notifier). Unit tests run use cases against in-memory fakes; integration tests hit the API over SQLite (StaticPool); runtime is PostgreSQL 16 + Alembic.

**Tech Stack:** Python 3.12 · FastAPI · Pydantic v2 · SQLAlchemy 2.0 · Alembic · PostgreSQL 16 · PyJWT · bcrypt · pytest/pytest-cov · flake8/black/isort/ruff · Docker multistage + compose.

**Binding conventions (from spec):** SOLID explicit; **no docstrings, no inline comments anywhere**; semantic names — functions read as the business rule; tests read as behavior specs.

from datetime import datetime, timezone
from uuid import UUID, uuid4

from pydantic import BaseModel, Field

from app.domain.enums import ActivityStatus, Priority, Role
from app.domain.exceptions import (
    PathHasNoActivitiesError,
    PathHasPendingMandatoryActivitiesError,
)


def utcnow() -> datetime:
    return datetime.now(timezone.utc)


class User(BaseModel):
    id: UUID = Field(default_factory=uuid4)
    email: str
    name: str
    role: Role
    hashed_password: str
    created_at: datetime = Field(default_factory=utcnow)


class Activity(BaseModel):
    id: UUID = Field(default_factory=uuid4)
    path_id: UUID
    title: str
    description: str | None = None
    priority: Priority = Priority.MEDIUM
    is_mandatory: bool = False
    status: ActivityStatus = ActivityStatus.NOT_STARTED
    assigned_to: UUID | None = None
    position: int = 1
    created_at: datetime = Field(default_factory=utcnow)
    updated_at: datetime = Field(default_factory=utcnow)


class LearningPath(BaseModel):
    id: UUID = Field(default_factory=uuid4)
    title: str
    description: str | None = None
    created_by: UUID
    completed_at: datetime | None = None
    created_at: datetime = Field(default_factory=utcnow)
    activities: list[Activity] = Field(default_factory=list)

    @property
    def activity_count(self) -> int:
        return len(self.activities)

    @property
    def progress_percentage(self) -> float:
        if not self.activities:
            return 0.0
        completed = sum(1 for a in self.activities if a.status is ActivityStatus.COMPLETED)
        return round(completed * 100 / len(self.activities), 2)

    @property
    def is_compliant(self) -> bool:
        return all(a.status is ActivityStatus.COMPLETED for a in self.activities if a.is_mandatory)

    def sign_off(self, at: datetime) -> None:
        if not self.activities:
            raise PathHasNoActivitiesError()
        if not self.is_compliant:
            raise PathHasPendingMandatoryActivitiesError()
        self.completed_at = at
```

from app.domain.entities import LearningPath, User
from app.domain.enums import Role
from app.domain.exceptions import PermissionDeniedError


def require_leader(actor: User) -> None:
    if actor.role is not Role.LEADER:
        raise PermissionDeniedError()


def ensure_owner(path: LearningPath, actor: User) -> None:
    if actor.role is not Role.LEADER or path.created_by != actor.id:
        raise PermissionDeniedError()


def ensure_can_view(path: LearningPath, actor: User) -> None:
    if actor.role is Role.LEADER and path.created_by == actor.id:
        return
    if any(a.assigned_to == actor.id for a in path.activities):
        return
    raise PermissionDeniedError()

from uuid import UUID

from app.domain.entities import Activity, LearningPath, User
from app.domain.ports import (
    ActivityRepository,
    LearningPathRepository,
    Notifier,
    PasswordHasher,
    UserRepository,
)


class FakeUserRepository(UserRepository):
    def __init__(self) -> None:
        self.users: dict[UUID, User] = {}

    def add(self, user: User) -> None:
        self.users[user.id] = user

    def get_by_id(self, user_id: UUID) -> User | None:
        return self.users.get(user_id)

    def get_by_email(self, email: str) -> User | None:
        return next((u for u in self.users.values() if u.email == email), None)


class FakeActivityRepository(ActivityRepository):
    def __init__(self) -> None:
        self.activities: dict[UUID, Activity] = {}

    def add(self, activity: Activity) -> None:
        self.activities[activity.id] = activity

    def get(self, activity_id: UUID) -> Activity | None:
        return self.activities.get(activity_id)

    def update(self, activity: Activity) -> None:
        self.activities[activity.id] = activity

    def delete(self, activity_id: UUID) -> None:
        self.activities.pop(activity_id, None)


class FakeLearningPathRepository(LearningPathRepository):
    def __init__(self, activity_repo: FakeActivityRepository) -> None:
        self.paths: dict[UUID, LearningPath] = {}
        self.activity_repo = activity_repo

    def add(self, path: LearningPath) -> None:
        self.paths[path.id] = path

    def get(self, path_id: UUID) -> LearningPath | None:
        path = self.paths.get(path_id)
        if path is None:
            return None
        path.activities = sorted(
            (a for a in self.activity_repo.activities.values() if a.path_id == path_id),
            key=lambda a: a.position,
        )
        return path

    def list_owned_by(self, leader_id: UUID) -> list[LearningPath]:
        return [self.get(p.id) for p in self.paths.values() if p.created_by == leader_id]

    def list_assigned_to(self, member_id: UUID) -> list[LearningPath]:
        assigned_path_ids = {
            a.path_id
            for a in self.activity_repo.activities.values()
            if a.assigned_to == member_id
        }
        return [self.get(pid) for pid in assigned_path_ids if pid in self.paths]

    def update(self, path: LearningPath) -> None:
        self.paths[path.id] = path

    def delete(self, path_id: UUID) -> None:
        self.paths.pop(path_id, None)
        for activity_id in [
            a.id for a in self.activity_repo.activities.values() if a.path_id == path_id
        ]:
            self.activity_repo.delete(activity_id)


class RecordingNotifier(Notifier):
    def __init__(self) -> None:
        self.invitations: list[User] = []
        self.assignments: list[tuple[User, Activity]] = []

    def send_invitation(self, user: User) -> None:
        self.invitations.append(user)

    def send_assignment(self, user: User, activity: Activity) -> None:
        self.assignments.append((user, activity))


class PlainTextHasher(PasswordHasher):
    def hash(self, plain: str) -> str:
        return f"hashed:{plain}"

    def verify(self, plain: str, hashed: str) -> bool:
        return hashed == f"hashed:{plain}"

from app.application.dtos import ProvisionMemberCommand
from app.domain.entities import User
from app.domain.enums import Role
from app.domain.exceptions import EmailAlreadyRegisteredError, InvalidCredentialsError
from app.domain.policies import require_leader
from app.domain.ports import Notifier, PasswordHasher, UserRepository


class AuthenticateUser:
    def __init__(self, users: UserRepository, hasher: PasswordHasher) -> None:
        self.users = users
        self.hasher = hasher

    def execute(self, email: str, password: str) -> User:
        user = self.users.get_by_email(email)
        if user is None or not self.hasher.verify(password, user.hashed_password):
            raise InvalidCredentialsError()
        return user


class ProvisionMember:
    def __init__(
        self, users: UserRepository, hasher: PasswordHasher, notifier: Notifier
    ) -> None:
        self.users = users
        self.hasher = hasher
        self.notifier = notifier

    def execute(self, actor: User, command: ProvisionMemberCommand) -> User:
        require_leader(actor)
        if self.users.get_by_email(command.email) is not None:
            raise EmailAlreadyRegisteredError()

            email=command.email,

def matches(activity: Activity, filters: ActivityFilters) -> bool:
    if filters.status is not None and activity.status is not filters.status:
        return False
    if filters.priority is not None and activity.priority is not filters.priority:
        return False
    return True


class AddActivity:
    def __init__(self, paths: LearningPathRepository, activities: ActivityRepository) -> None:
        self.paths = paths
        self.activities = activities

    def execute(self, actor: User, path_id: UUID, command: AddActivityCommand) -> Activity:
        path = load_path(self.paths, path_id)
        ensure_owner(path, actor)
        activity = Activity(
            path_id=path.id,
            title=command.title,
            description=command.description,
            priority=command.priority,
            is_mandatory=command.is_mandatory,
            position=path.activity_count + 1,
        )
        self.activities.add(activity)
        return activity


class ListActivities:
    def __init__(self, paths: LearningPathRepository) -> None:
        self.paths = paths

    def execute(self, actor: User, path_id: UUID, filters: ActivityFilters) -> ActivityListResult:
        path = load_path(self.paths, path_id)
        ensure_can_view(path, actor)
        filtered = [a for a in path.activities if matches(a, filters)]
        return ActivityListResult(
            activities=filtered, progress_percentage=path.progress_percentage
        )


class UpdateActivity:
    def __init__(self, paths: LearningPathRepository, activities: ActivityRepository) -> None:
        self.paths = paths
        self.activities = activities

    def execute(self, actor: User, activity_id: UUID, command: UpdateActivityCommand) -> Activity:
        activity = load_activity(self.activities, activity_id)
        path = load_path(self.paths, activity.path_id)
        ensure_owner(path, actor)
        for field in ("title", "description", "priority", "is_mandatory", "position"):
            value = getattr(command, field)
            if value is not None:
                setattr(activity, field, value)
        activity.updated_at = utcnow()
        self.activities.update(activity)
        return activity


class UpdateActivityStatus:
    def __init__(self, paths: LearningPathRepository, activities: ActivityRepository) -> None:
        self.paths = paths
        self.activities = activities

    def execute(self, actor: User, activity_id: UUID, status: ActivityStatus) -> Activity:
        activity = load_activity(self.activities, activity_id)
        path = load_path(self.paths, activity.path_id)
        if not self._actor_may_change_status(actor, path, activity):
            raise PermissionDeniedError()
        activity.status = status
        activity.updated_at = utcnow()
        self.activities.update(activity)
        return activity

    @staticmethod
    def _actor_may_change_status(actor: User, path, activity: Activity) -> bool:
        owns_path = path.created_by == actor.id
        is_assignee = activity.assigned_to == actor.id
        return owns_path or is_assignee

def activity_to_entity(row: ActivityModel) -> Activity:
    return Activity.model_validate(row, from_attributes=True)


def path_to_entity(row: LearningPathModel) -> LearningPath:
    path = LearningPath.model_validate(row, from_attributes=True)
    path.activities = [activity_to_entity(a) for a in row.activities]
    return path


class SqlAlchemyUserRepository(UserRepository):
    def __init__(self, session: Session) -> None:
        self.session = session

    def add(self, user: User) -> None:
        self.session.add(UserModel(**user.model_dump()))
        self.session.flush()

    def get_by_id(self, user_id: UUID) -> User | None:
        row = self.session.get(UserModel, user_id)
        return user_to_entity(row) if row else None

    def get_by_email(self, email: str) -> User | None:
        row = self.session.scalar(select(UserModel).where(UserModel.email == email))
        return user_to_entity(row) if row else None


class SqlAlchemyLearningPathRepository(LearningPathRepository):
    def __init__(self, session: Session) -> None:
        self.session = session

    def add(self, path: LearningPath) -> None:
        self.session.add(LearningPathModel(**path.model_dump(exclude={"activities"})))
        self.session.flush()

    def get(self, path_id: UUID) -> LearningPath | None:
        row = self.session.scalar(
            select(LearningPathModel)
            .options(selectinload(LearningPathModel.activities))
            .where(LearningPathModel.id == path_id)
        )
        return path_to_entity(row) if row else None

    def list_owned_by(self, leader_id: UUID) -> list[LearningPath]:
        rows = self.session.scalars(
            select(LearningPathModel)
            .options(selectinload(LearningPathModel.activities))
            .where(LearningPathModel.created_by == leader_id)
            .order_by(LearningPathModel.created_at)
        )
        return [path_to_entity(r) for r in rows]

    def list_assigned_to(self, member_id: UUID) -> list[LearningPath]:
        rows = self.session.scalars(
            select(LearningPathModel)
            .options(selectinload(LearningPathModel.activities))
            .join(ActivityModel, ActivityModel.path_id == LearningPathModel.id)
            .where(ActivityModel.assigned_to == member_id)
            .distinct()
            .order_by(LearningPathModel.created_at)
        )
        return [path_to_entity(r) for r in rows]

    def update(self, path: LearningPath) -> None:
        row = self.session.get(LearningPathModel, path.id)
        row.title = path.title
        row.description = path.description
        row.completed_at = path.completed_at
        self.session.flush()

    def delete(self, path_id: UUID) -> None:
        row = self.session.get(LearningPathModel, path_id)
        if row is not None:
            self.session.delete(row)
            self.session.flush()

        self.session.add(ActivityModel(**activity.model_dump()))
        self.session.flush()

    def get(self, activity_id: UUID) -> Activity | None:
        row = self.session.get(ActivityModel, activity_id)
        return activity_to_entity(row) if row else None

    def update(self, activity: Activity) -> None:
        row = self.session.get(ActivityModel, activity.id)
        for field, value in activity.model_dump(exclude={"id"}).items():
            setattr(row, field, value)
        self.session.flush()

    def delete(self, activity_id: UUID) -> None:
        row = self.session.get(ActivityModel, activity_id)
        if row is not None:
            self.session.delete(row)
            self.session.flush()
```

from datetime import timedelta

import bcrypt
import jwt

from app.domain.entities import User, utcnow
from app.domain.ports import PasswordHasher
from app.infrastructure.config import settings

ALGORITHM = "HS256"


class BcryptPasswordHasher(PasswordHasher):
    def __init__(self, rounds: int = 12) -> None:
        self.rounds = rounds

    def hash(self, plain: str) -> str:
        return bcrypt.hashpw(plain.encode(), bcrypt.gensalt(rounds=self.rounds)).decode()

    def verify(self, plain: str, hashed: str) -> bool:
        return bcrypt.checkpw(plain.encode(), hashed.encode())


def create_access_token(user: User) -> str:
    payload = {
        "sub": str(user.id),
        "exp": utcnow() + timedelta(minutes=settings.jwt_expires_minutes),
    }
    return jwt.encode(payload, settings.jwt_secret, algorithm=ALGORITHM)


def decode_token(token: str) -> str:
    payload = jwt.decode(token, settings.jwt_secret, algorithms=[ALGORITHM])
    return payload["sub"]
```

    credentials: HTTPAuthorizationCredentials = Depends(bearer_scheme),
    users: SqlAlchemyUserRepository = Depends(get_user_repository),
) -> User:
    try:
        user_id = decode_token(credentials.credentials)

from app.domain.entities import Activity, LearningPath, User
from app.domain.enums import ActivityStatus, Priority, Role
from app.infrastructure.auth.security import BcryptPasswordHasher
from app.infrastructure.db.repositories import (
    SqlAlchemyActivityRepository,
    SqlAlchemyLearningPathRepository,
    SqlAlchemyUserRepository,
)
from app.infrastructure.db.session import SessionLocal

DEMO_USERS = [
    ("leader@learnflow.dev", "Lia Leader", Role.LEADER, "leader-pass"),
    ("member1@learnflow.dev", "Max Member", Role.MEMBER, "member-pass"),
    ("member2@learnflow.dev", "Mia Member", Role.MEMBER, "member-pass"),
]

DEMO_ACTIVITIES = [
    ("Security and compliance onboarding", Priority.HIGH, True, ActivityStatus.COMPLETED),
    ("Git workflow and code review basics", Priority.MEDIUM, True, ActivityStatus.NOT_STARTED),
    ("FastAPI fundamentals course", Priority.HIGH, False, ActivityStatus.IN_PROGRESS),
    ("Team wiki deep dive", Priority.LOW, False, ActivityStatus.NOT_STARTED),
]


def ensure_users(users: SqlAlchemyUserRepository, hasher: BcryptPasswordHasher) -> dict[str, User]:
    by_email = {}
    for email, name, role, password in DEMO_USERS:
        existing = users.get_by_email(email)
        if existing is None:
            existing = User(email=email, name=name, role=role, hashed_password=hasher.hash(password))
            users.add(existing)
            print(f"seeded {role}: {email} / {password}")
        by_email[email] = existing
    return by_email


def ensure_demo_path(
    paths: SqlAlchemyLearningPathRepository,
    activities: SqlAlchemyActivityRepository,
    leader: User,
    member: User,
) -> None:
    if paths.list_owned_by(leader.id):
        return
    path = LearningPath(
        title="Backend onboarding",
        description="First-quarter ramp-up for new backend engineers",
        created_by=leader.id,
    )
    paths.add(path)
    for position, (title, priority, is_mandatory, status) in enumerate(DEMO_ACTIVITIES, start=1):
        activities.add(
            Activity(
                path_id=path.id,
                title=title,
                priority=priority,
                is_mandatory=is_mandatory,
                status=status,
                assigned_to=member.id,
                position=position,
            )
        )
    print(f"seeded path: {path.title} ({len(DEMO_ACTIVITIES)} activities, assigned to {member.email})")


def run() -> None:
    hasher = BcryptPasswordHasher()
    with SessionLocal() as session:
        users = ensure_users(SqlAlchemyUserRepository(session), hasher)
        ensure_demo_path(
            SqlAlchemyLearningPathRepository(session),
            SqlAlchemyActivityRepository(session),
            users["leader@learnflow.dev"],
            users["member1@learnflow.dev"],
        )
        session.commit()


if __name__ == "__main__":
    run()
```

Demo path state on purpose: 25% progress (1 of 4 completed), **not compliant** (mandatory
"Git workflow" pending) — sign-off blocked until the member finishes it; the two optional
activities never block. Shows compliance ≠ completion without any clicks. Idempotent: skips
when the leader already owns a path.
